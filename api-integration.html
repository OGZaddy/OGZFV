Quick Start Code Examples

JavaScript / Node.js

// Get Latest Signals
const apiKey = 'YOUR_API_KEY_HERE';

fetch('https://ogzprime.com/api/signals/latest', {
    headers: {
        'Authorization': `Bearer ${apiKey}`
    }
})
.then(response => response.json())
.then(data => {
    console.log('Latest signals:', data.signals);
    data.signals.forEach(signal => {
        console.log(`${signal.action} ${signal.symbol} at $${signal.price} (${signal.confidence}% confidence)`);
    });
})
.catch(error => console.error('Error:', error));

WebSocket Real-Time (Premium+)

// Real-time WebSocket Connection
const WebSocket = require('ws');
const apiKey = 'YOUR_API_KEY_HERE';

const ws = new WebSocket('wss://ogzprime.com/api/realtime', {
    headers: { 'Authorization': `Bearer ${apiKey}` }
});

ws.on('open', () => {
    console.log('🚀 Connected to OGZPrime real-time signals');
});

ws.on('message', (data) => {
    const signal = JSON.parse(data);
    
    if (signal.type === 'trade_signal') {
        console.log(`🔔 NEW SIGNAL: ${signal.action} ${signal.symbol} at $${signal.price}`);
        console.log(`   Confidence: ${signal.confidence}%`);
        console.log(`   Reasoning: ${signal.reasoning}`);
        
        // Your trading logic here
        executeTradeSignal(signal);
    }
});

ws.on('error', (error) => {
    console.error('WebSocket error:', error);
});

Python Example

import requests
import json

# Get Latest Signals
api_key = 'YOUR_API_KEY_HERE'
headers = {'Authorization': f'Bearer {api_key}'}

response = requests.get('https://ogzprime.com/api/signals/latest', headers=headers)
data = response.json()

for signal in data['signals']:
    print(f"{signal['action']} {signal['symbol']} at ${signal['price']} ({signal['confidence']}% confidence)")
    
# Get Performance Metrics (Premium+)
perf_response = requests.get('https://ogzprime.com/api/performance', headers=headers)
metrics = perf_response.json()
print(f"Total signals (30 days): {metrics['metrics']['total_signals']}")
print(f"Average confidence: {metrics['metrics']['average_confidence']}%")
