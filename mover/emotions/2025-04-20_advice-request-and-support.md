function getTrend(data, shortWindow = 9, longWindow = 21) {
    if (!data || data.length < longWindow) return null;

    const shortMA = data.slice(-shortWindow).reduce((sum, d) => sum + d.close, 0) / shortWindow;
    const longMA = data.slice(-longWindow).reduce((sum, d) => sum + d.close, 0) / longWindow;

    if (shortMA > longMA) return 'uptrend';

    return 'sideways';
}

// claudeBrain.js
// CLAUDE BRAIN MODULE 1 - TREND CONFIRMATION LOGIC
// This module analyzes market conditions to determine trend direction
// and filters trade signals based on momentum alignment.

function getTrend(data, shortWindow = 9, longWindow = 21) {
    if (!data || data.length < longWindow) return null;

    const shortMA = data.slice(-shortWindow).reduce((sum, d) => sum + d.close, 0) / shortWindow;
    const longMA = data.slice(-longWindow).reduce((sum, d) => sum + d.close, 0) / longWindow;

    if (shortMA > longMA) return 'uptrend';
    if (shortMA < longMA) return 'downtrend';
    return 'sideways';
}

function filterSignalByTrend(signal, trend) {
    if (trend === 'uptrend' && signal === 'buy') return true;
    if (trend === 'downtrend' && signal === 'sell') return true;
    if (trend === 'sideways') return false;
    return false;
}

function analyzeAndDecide(candles, rawSignal) {
    const trend = getTrend(candles);
    const approved = filterSignalByTrend(rawSignal, trend);

    return {
        trend,
        rawSignal,
        decision: approved ? rawSignal : 'hold',
        log: `Trend: ${trend} | Signal: ${rawSignal} => Decision: ${approved ? rawSignal : 'HOLD'}`
    };
}

console.log("[DEBUG] claudeBrain.js module successfully loaded");

module.exports = {

};

function aggregateToCandle() {
    if (trades.length === 0) return;

    const currentTime = trades[trades.length - 1].time;

    if (lastCandleTime === null) {
        lastCandleTime = currentTime;
    }

    if (currentTime !== lastCandleTime) {
        const candleTrades = trades.filter(t => t.time === lastCandleTime);
        const prices = candleTrades.map(t => t.price);

        if (prices.length > 0) {
            const candle = {
                open: prices[0],
                high: Math.max(...prices),
                low: Math.min(...prices),
                close: prices[prices.length - 1]

                executeSell(candle.close);
            } else {
                console.log("[ClaudeBrain] HOLD. No trade executed.");
            }
        }

        // Purge old trades
        trades = trades.filter(t => t.time > lastCandleTime);
        lastCandleTime = currentTime;

    fs.appendFileSync('pnl_log.txt', log);
}

// CLAUDE BRAIN MODULE V2 - TREND + RSI-BASED DECISION ENGINE

function getTrend(data, shortWindow = 9, longWindow = 21) {
    if (!data || data.length < longWindow) return null;

    const shortMA = data.slice(-shortWindow).reduce((sum, d) => sum + d.close, 0) / shortWindow;
    const longMA = data.slice(-longWindow).reduce((sum, d) => sum + d.close, 0) / longWindow;

    if (shortMA > longMA) return 'uptrend';
    if (shortMA < longMA) return 'downtrend';
    return 'sideways';
}

function filterSignalByTrend(signal, trend) {
    if (trend === 'uptrend' && signal === 'buy') return true;
    if (trend === 'downtrend' && signal === 'sell') return true;
    return false;
}

function calculateRSI(candles, period = 14) {
    if (candles.length < period + 1) return null;

    let gains = 0;
    let losses = 0;

    for (let i = candles.length - period; i < candles.length; i++) {
        const change = candles[i].close - candles[i - 1].close;
        if (change >= 0) gains += change;
        else losses -= change;
    }

    if (gains === 0) return 0;
    if (losses === 0) return 100;

    const rs = gains / losses;
    return 100 - (100 / (1 + rs));
}

function generateSmartSignal(candles) {
    const rsi = calculateRSI(candles);
    const trend = getTrend(candles);

    if (!rsi || !trend) return 'hold';

    if (rsi < 30 && trend === 'uptrend') return 'buy';
    if (rsi > 70 && trend === 'downtrend') return 'sell';

    return 'hold';
}

function analyzeAndDecide(candles, rawSignal) {
    const trend = getTrend(candles);
    const approved = filterSignalByTrend(rawSignal, trend);

    return {
        trend,
        rawSignal,
        decision: approved ? rawSignal : 'hold',
        log: `Trend: ${trend} | Signal: ${rawSignal} => Decision: ${approved ? rawSignal : 'HOLD'}`
    };
}

console.log("[DEBUG] claudeBrain.js module successfully loaded");

    getTrend,
    filterSignalByTrend,
    analyzeAndDecide,
    calculateRSI,
    generateSmartSignal
};

function getTrend(data, shortWindow = 9, longWindow = 21) {
    if (!data || data.length < longWindow) return null;

    const shortMA = data.slice(-shortWindow).reduce((sum, d) => sum + d.close, 0) / shortWindow;
    const longMA = data.slice(-longWindow).reduce((sum, d) => sum + d.close, 0) / longWindow;

    if (shortMA > longMA) return 'uptrend';

function calculateRSI(candles, period = 14) {
    if (candles.length < period + 1) return null;

    let gains = 0;
    let losses = 0;

    for (let i = candles.length - period; i < candles.length; i++) {
        const change = candles[i].close - candles[i - 1].close;
        if (change >= 0) gains += change;
        else losses -= change;
    }

    if (gains === 0) return 0;
    if (losses === 0) return 100;

    const rs = gains / losses;
    return 100 - (100 / (1 + rs));
}

function filterSignalByTrend(signal, trend) {
