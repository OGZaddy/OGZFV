## 🔧 ✅ Fix: Use Relative Hostname Dynamically

### 👉 Replace all WebSocket URLs like this:

- On localhost (`localhost:3010`)
- On VPS IP (`123.45.67.89:3010`)
- On domain (`ogzprime.com:3010`)


/**

 * 
 * SIMULATION PURPOSE:
 * Projects potential future performance based on historical trading patterns
 * to estimate probability of reaching Houston move financial goals.
 * 
 * @param {number} currentBalance - Current account balance
 * @param {number} targetAmount - Target amount for Houston move
 * @param {number} timeHorizon - Days to simulate forward
 * @param {number} iterations - Number of simulation runs
 * @returns {Object} Simulation results with confidence intervals
 */
window.runMonteCarloSimulation = function(currentBalance, targetAmount, timeHorizon = 365, iterations = 1000) {
  console.log('📊 Running Monte Carlo simulation for Houston goal projection...');
  
  // Historical performance metrics (would be populated from real data)
  const historicalReturns = getHistoricalDailyReturns();
  const avgDailyReturn = historicalReturns.mean || 0.001; // 0.1% daily average
  const volatility = historicalReturns.stdDev || 0.02; // 2% daily volatility
  
  const results = [];
  let successCount = 0;
  
  // Run Monte Carlo iterations
  for (let i = 0; i < iterations; i++) {
    let balance = currentBalance;
    
    // Simulate daily returns for time horizon
    for (let day = 0; day < timeHorizon; day++) {
      // Generate random return using normal distribution
      const randomReturn = generateNormalRandom(avgDailyReturn, volatility);
      balance *= (1 + randomReturn);
      
      // Stop early if target reached

        successCount++;
        break;
      }
    }
    
    results.push(balance);
  }
  
  // Calculate statistics

  const successProbability = (successCount / iterations) * 100;
  
  const simulation = {
    successProbability: successProbability,
    median: results[Math.floor(iterations * 0.5)],
    confidence95: results[Math.floor(iterations * 0.95)],
    confidence5: results[Math.floor(iterations * 0.05)],
    targetAmount: targetAmount,
    timeHorizon: timeHorizon,
    currentBalance: currentBalance
  };
  
  // Update Monte Carlo display
  updateMonteCarloDisplay(simulation);
  
  return simulation;
};

/**
 * Generate normally distributed random number
 * 
 * @param {number} mean - Mean of distribution
 * @param {number} stdDev - Standard deviation
 * @returns {number} Random number from normal distribution
 */
function generateNormalRandom(mean, stdDev) {
  // Box-Muller transformation for normal distribution
  const u1 = Math.random();
  const u2 = Math.random();
  const z0 = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  return mean + stdDev * z0;
}
