/**
 * Calculate market volatility
 * @param {Array} candles - Recent market candles
 * @param {number} windowSize - Number of candles to analyze
 * @returns {number} Volatility measure
 */
function calculateVolatility(candles, windowSize) {
  // Take last 'windowSize' candles
  const recentCandles = candles.slice(-windowSize);
  
  // Calculate percentage changes
  const changes = recentCandles.map((candle, index) => {
    if (index === 0) return 0;
    return (candle.close - recentCandles[index-1].close) / recentCandles[index-1].close;
  });
  
  // Calculate standard deviation of changes
  const mean = changes.reduce((a, b) => a + b, 0) / changes.length;
  const variance = changes.reduce((sq, n) => sq + Math.pow(n - mean, 2), 0) / changes.length;
  
  return Math.sqrt(variance);
}
