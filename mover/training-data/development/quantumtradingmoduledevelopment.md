// Smart indicator caching with volatility-adjusted expiration
class IndicatorCache {
  constructor() {
    this.cache = new Map();
    this.baseExpiry = 30000; // 30 seconds base
  }
  
  calculateExpiry(volatility) {
    // Higher volatility = shorter cache life
    return this.baseExpiry * Math.max(0.1, 1 - volatility * 2);
  }
  
  get(indicator, timeframe, volatility) {
    const key = `${indicator}_${timeframe}`;
    const cached = this.cache.get(key);
    
    const dynamicExpiry = this.calculateExpiry(volatility);
    
    if (cached && (Date.now() - cached.timestamp < dynamicExpiry)) {
      return cached.value; // Fresh enough
    }
    
    return null; // Expired, recalculate
  }
}
