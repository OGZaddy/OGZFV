```jsx
import React, { useEffect, useState } from "react";
import { Line } from "react-chartjs-2";
import axios from "axios";
import { Chart as ChartJS, CategoryScale, LinearScale, PointElement, LineElement, Title, Tooltip, Legend } from "chart.js";

// Register Chart.js components
ChartJS.register(CategoryScale, LinearScale, PointElement, LineElement, Title, Tooltip, Legend);

const StockChart = () => {
  const [chartData, setChartData] = useState({
    labels: [],
    datasets: [
      {
        label: "Stock Price",
        data: [],
        borderColor: "blue",
        backgroundColor: "rgba(0, 0, 255, 0.5)",
        tension: 0.4,
      },
    ],
  });

  useEffect(() => {
    const fetchData = async () => {
      try {
        // Replace 'AAPL' with any stock ticker you want
        const response = await axios.get(
          "https://www.alphavantage.co/query?function=TIME_SERIES_INTRADAY&symbol=AAPL&interval=5min&apikey=demo"
        );
        
        const stockData = response.data["Time Series (5min)"];
        const labels = Object.keys(stockData).reverse();
        const prices = labels.map((timestamp) => stockData[timestamp]["1. open"]);

        setChartData({
          labels,
          datasets: [
            {
              label: "AAPL Stock Price",
              data: prices,
              borderColor: "blue",
              backgroundColor: "rgba(0, 0, 255, 0.5)",
              tension: 0.4,
            },
          ],
        });
      } catch (error) {
        console.error("Error fetching stock data:", error);
      }
    };

    fetchData();
    const interval = setInterval(fetchData, 60000); // Refresh every 60 seconds

    return () => clearInterval(interval);
  }, []);

  return (
    <div>
      <h2>Stock Price Chart (AAPL)</h2>
      <Line data={chartData} />
    </div>
  );
};

export default StockChart;
```
