## 1. StochasticOscillator.js  
_Existing oscillator code wrapped into a module for easy plug-in._

```javascript
// 📁 core/StochasticOscillator.js
// “Beat the crowd by knowing overbought/oversold before they do.”

class StochasticOscillator {
  constructor(config = {}) {
    this.kPeriod = config.kPeriod || 14;
    this.dPeriod = config.dPeriod || 3;
    this.smoothing = config.smoothing || 3;
    
    this._kValues = [];
    this._dValues = [];
  }

  /**
   * Feed one new candle: { high, low, close }.
   * Returns { percentK, percentD } for this point.
   */
  update(candle) {
    this._kValues.push({ high: candle.high, low: candle.low, close: candle.close });
    if (this._kValues.length < this.kPeriod) {
      return { percentK: null, percentD: null };
    }

    // Slice the last kPeriod candles
    const slice = this._kValues.slice(-this.kPeriod);
    const highs = slice.map(c => c.high);
    const lows = slice.map(c => c.low);
    const highestHigh = Math.max(...highs);
    const lowestLow = Math.min(...lows);
    
    const percentK = ((candle.close - lowestLow) / (highestHigh - lowestLow)) * 100;
    this._dValues.push(percentK);

    // Smooth %D
    if (this._dValues.length < this.dPeriod) {
      return { percentK, percentD: null };
    }
    const dSlice = this._dValues.slice(-this.dPeriod);
    const percentD = dSlice.reduce((sum, v) => sum + v, 0) / dSlice.length;

    return { percentK, percentD };
  }

  reset() {
    this._kValues = [];
    this._dValues = [];
  }
}

module.exports = StochasticOscillator;
```

## 2. DataCompressionModule.js  
_Compress old pattern data files (e.g., `data/pattern-memory.json`) to save disk and network bandwidth._

```javascript
// 📁 core/DataCompressionModule.js
// “Archive ancient patterns—only keep the hot ones uncompressed.”

const fs = require('fs');
const path = require('path');
const zlib = require('zlib');

class DataCompressionModule {
  constructor(options = {}) {
    this.patternDir = options.patternDir || path.join(__dirname, '../data');
    this.thresholdDays = options.thresholdDays || 30;
  }

  /**
   * Compress all JSON files older than `thresholdDays`.
   */
  compressOldPatterns() {
    const cutoff = Date.now() - this.thresholdDays * 24 * 60 * 60 * 1000;
    const files = fs.readdirSync(this.patternDir);

    files.forEach(file => {
      if (!file.endsWith('.json')) return;
      const fullPath = path.join(this.patternDir, file);
      const stat = fs.statSync(fullPath);
      if (stat.mtimeMs < cutoff && !file.endsWith('.gz')) {
        const dest = fullPath + '.gz';
        const input = fs.createReadStream(fullPath);
        const output = fs.createWriteStream(dest);
        input.pipe(zlib.createGzip()).pipe(output);
        output.on('finish', () => {
          fs.unlinkSync(fullPath);
          console.log(`📦 Compressed and archived: ${file}`);
        });
      }
    });
  }

  /**
   * Decompress a specific file if needed.
   */
  decompressPattern(compressedFilename) {
    const gzPath = path.join(this.patternDir, compressedFilename);
    if (!gzPath.endsWith('.gz') || !fs.existsSync(gzPath)) return false;

    const destPath = gzPath.replace(/\.gz$/, '');
    const input = fs.createReadStream(gzPath);
    const output = fs.createWriteStream(destPath);
    input.pipe(zlib.createGunzip()).pipe(output);
    output.on('finish', () => {
      console.log(`📂 Decompressed: ${compressedFilename}`);
    });
    return true;
  }
}

module.exports = DataCompressionModule;
```

## 4. CustomAlertsPanel.js  
_An in-app panel for alerts, beyond Discord—supports priority levels and voice cues._

```javascript
// 📁 core/CustomAlertsPanel.js
// “When Discord sleeps, this wakes you up—red alerts, battle cries, and victory jingles.”

class CustomAlertsPanel {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    if (!this.container) {
      console.warn(`⚠️ No element with ID "${containerId}" found for alerts.`);
      return;
    }
    this.alerts = [];
  }

  /**
   * Create a new alert.
   * @param {string} message - Text to show.
   * @param {'info'|'warning'|'critical'|'victory'} level - Severity.
   * @param {Array<{label:string, action:Function}>} actions - Optional buttons.
   */
  createAlert(message, level = 'info', actions = []) {
    if (!this.container) return;
    const alertEl = document.createElement('div');
    alertEl.classList.add('custom-alert', `alert-${level}`);
    alertEl.innerHTML = `
      <p>${message}</p>
      <div class="alert-actions"></div>
      <span class="alert-close">×</span>
    `;
    const actionContainer = alertEl.querySelector('.alert-actions');
    actions.forEach(act => {
      const btn = document.createElement('button');
      btn.textContent = act.label;
      btn.className = 'alert-btn';
      btn.addEventListener('click', () => act.action());
      actionContainer.appendChild(btn);
    });
    // Handle close
    alertEl.querySelector('.alert-close').addEventListener('click', () => {
      this.container.removeChild(alertEl);
    });
    this.container.appendChild(alertEl);
    this.alerts.push(alertEl);

    // Voice cue for critical/victory
    if (window.speechSynthesis && (level === 'critical' || level === 'victory')) {
      const utterance = new SpeechSynthesisUtterance(message);
      utterance.rate = 1.2;
      utterance.pitch = level === 'victory' ? 1.5 : 0.8;
      speechSynthesis.speak(utterance);
    }
  }

  clearAll() {
    this.alerts.forEach(el => el.remove());
    this.alerts = [];
  }
}

module.exports = CustomAlertsPanel;
```

> **Usage:** Add a `<div id="alerts-container"></div>` in your HTML.  
> Call `new CustomAlertsPanel('alerts-container')` in your front-end code.

## 5. AutoCloseOnDisconnect.js  
_Ensure any open position is closed if the bot loses connection to the exchange._

```javascript
// 📁 core/AutoCloseOnDisconnect.js
// “No endless hung orders—cut positions if the link dies.”

class AutoCloseOnDisconnect {
  constructor(tradingBrain, options = {}) {
    this.tradingBrain = tradingBrain; // Your trading engine instance
    this.checkInterval = options.checkInterval || 5000; // 5 seconds
    this._startMonitoring();
  }

  _startMonitoring() {
    this.monitor = setInterval(() => {
      if (!this.tradingBrain.isConnected()) {
        console.warn('⚠️ Disconnected from exchange! Closing any open positions...');
        if (this.tradingBrain.isInPosition()) {
          this.tradingBrain.executeManualSell(); // Force market‐sell or close logic
        }
        // Optionally: Attempt reconnection
        this.tradingBrain.reconnect().catch(err => {
          console.error('🔄 Reconnect failed:', err.message);
        });
      }
    }, this.checkInterval);
  }

  stop() {
    clearInterval(this.monitor);
  }
}

module.exports = AutoCloseOnDisconnect;
```

> **Usage:** Instantiate after your bot’s `tradingBrain` is initialized.  
> Requires your `tradingBrain` to have:  
> • `isConnected()` → bool  
> • `isInPosition()` → bool  
> • `executeManualSell()` → closes current position  
> • `reconnect()` → attempts to re‐open the connection

```javascript
// 📁 core/MobileResponsive.js
// “When you need Houston updates on the toilet at 3 AM—UI must fit your pocket.”

class MobileResponsive {
  constructor() {
    this._addMetaViewport();
    this._attachResizeListener();
  }

  _addMetaViewport() {
```

## 7. RealTimeSync.js  
_Keep pattern memory & config in sync between bot and remote monitor._

```javascript
// 📁 core/RealTimeSync.js
// “Your Houston phone sees exactly what your server sees—no stale data.”

const ws = require('ws');

class RealTimeSync {
  constructor(botInstance, options = {}) {
    this.bot = botInstance;
    this.port = options.port || 4050;
    this.clients = [];
    this._startServer();
  }

  _startServer() {
    this.wss = new ws.Server({ port: this.port });
    this.wss.on('connection', socket => {
      this.clients.push(socket);
      // Send initial state
      socket.send(JSON.stringify({ type: 'INIT', data: this._gatherBotState() }));
      socket.on('close', () => {
        this.clients = this.clients.filter(c => c !== socket);
      });
    });

    console.log(`🔗 RealTimeSync: WebSocket listening on port ${this.port}`);
  }

  _gatherBotState() {
    return {
      balance: this.bot.tradingBrain.balance,
      positions: this.bot.tradingBrain.position,
      config: this.bot.config,
      metrics: this.bot.status, // any additional status info
    };
  }

  broadcastUpdate(type, payload) {
    const message = JSON.stringify({ type, data: payload });
    this.clients.forEach(client => {
      if (client.readyState === ws.OPEN) {
        client.send(message);
      }
    });
  }

  /**
   * Call this whenever the bot’s state changes.
   * For example, after every trade or config adjustment:
   */
  onBotStateChange() {
    const update = this._gatherBotState();
    this.broadcastUpdate('UPDATE', update);
  }

  stop() {
    this.wss.close();
  }
}

module.exports = RealTimeSync;
```

```javascript
const path = require('path');
const PDFDocument = require('pdfkit'); // npm install pdfkit

class MonthlyReportGenerator {
  constructor(botInstance, options = {}) {
    this.bot = botInstance;

      const date = new Date(t.timestamp);
      return date.getUTCFullYear() === year && (date.getUTCMonth() + 1) === month;

    // Calculate summary metrics
    const totalPnL = trades.reduce((sum, t) => sum + t.pnl, 0);

    const losses = trades.filter(t => t.pnl <= 0).length;
    const winRate = trades.length ? (wins / trades.length) * 100 : 0;
    const bestTrade = trades.reduce((max, t) => (t.pnl > max.pnl ? t : max), { pnl: -Infinity });
    const worstTrade = trades.reduce((min, t) => (t.pnl < min.pnl ? t : min), { pnl: Infinity });

    const filepath = path.join(this.reportsDir, filename);
    const doc = new PDFDocument({ margin: 50 });
    doc.pipe(fs.createWriteStream(filepath));

    // Header
    doc.fontSize(20).text(`OGZ Prime Monthly Report: ${year}-${String(month).padStart(2,'0')}`, { align: 'center' });
    doc.moveDown();

    // Summary
    doc.fontSize(12).text(`Total Trades: ${trades.length}`);
    doc.text(`Wins: ${wins} | Losses: ${losses} | Win Rate: ${winRate.toFixed(1)}%`);
    doc.text(`Total PnL: $${totalPnL.toFixed(2)}`);
    doc.text(`Best Trade: $${bestTrade.pnl.toFixed(2)} on ${new Date(bestTrade.timestamp).toUTCString()}`);
    doc.text(`Worst Trade: $${worstTrade.pnl.toFixed(2)} on ${new Date(worstTrade.timestamp).toUTCString()}`);

    // Table of trades (first 10)
    doc.fontSize(10).text('Sample Trades:', { underline: true });
    const sample = trades.slice(0, 10);

    });

    // Footer
    doc.moveDown();
    doc.fontSize(10).text('Report generated by OGZ Prime', { align: 'center' });
```

```javascript
// 📁 core/CPUAndNetworkOptimization.js
// “More trades, fewer cycles—stop cooking your CPU and your data plan.”

class CPUAndNetworkOptimization {
  constructor(botInstance, options = {}) {
    this.bot = botInstance;
    this.maxCPUUsage = options.maxCPUUsage || 70; // percent
    this.minLoopInterval = options.minLoopInterval || 100; // ms
    this.lastLoopTime = 0;

    // Wrap the main analysis loop

        return;
      }
      this.lastLoopTime = now;
      await originalLoop(...args);
    };

  /**
   * Compress a JSON payload before sending over network.
   * Returns a Base64‐encoded gzip string.
   */
  static compressPayload(obj) {
    const json = JSON.stringify(obj);
    const buffer = Buffer.from(json, 'utf8');
    const compressed = require('zlib').gzipSync(buffer);
    return compressed.toString('base64');
  }

  /**
   * Decompress a Base64 gzip string back to object.
   */
  static decompressPayload(base64Str) {
    const buffer = Buffer.from(base64Str, 'base64');
    const decompressed = require('zlib').gunzipSync(buffer);
    return JSON.parse(decompressed.toString('utf8'));
  }
}
```

```javascript
const os = require('os');
const { exec } = require('child_process');

class SystemHealthMonitor {
  constructor(options = {}) {
    this.cpuThreshold = options.cpuThreshold || 85;    // %
    this.memThreshold = options.memThreshold || 80;    // %
    this.checkInterval = options.checkInterval || 60000; // 1 minute
    this.restartCommand = options.restartCommand || 'pm2 restart ogz-prime'; // or any systemctl/docker command

    this._startMonitoring();
  }

  _startMonitoring() {
    this.timer = setInterval(() => this._check(), this.checkInterval);
  }

  async _check() {
    const memUsage = this._getMemoryUsage();
    const cpuLoad = this._getCpuLoad();

    console.log(`🩺 Health: CPU ${cpuLoad.toFixed(1)}% | MEM ${memUsage.toFixed(1)}%`);

    if (cpuLoad > this.cpuThreshold || memUsage > this.memThreshold) {
      console.warn('🚨 Resource threshold exceeded! Initiating auto-restart...');
      this._restart();
    }
  }

  _getMemoryUsage() {
    const free = os.freemem();
    const total = os.totalmem();
    return ((1 - free / total) * 100);
  }

  _getCpuLoad() {
    // Average CPU load over 1 minute, scaled to percentage
    const load = os.loadavg()[0]; // load over last 1 min
    const cpus = os.cpus().length;
    return (load / cpus) * 100;
  }

  _restart() {
    exec(this.restartCommand, (err, stdout, stderr) => {
      if (err) {
        console.error(`❌ Restart failed: ${err.message}`);
        return;
      }
      console.log(`🔄 Restart command output: ${stdout}`);
    });
  }

  stop() {
    clearInterval(this.timer);
  }
}
```

## 15. BackupAutomation.js  
_Scheduled backups for configs, trade history, and pattern memory (compressing them)._

```javascript
// 📁 core/BackupAutomation.js
// “If the code, logs, or patterns disappear—rest easy knowing there’s a snapshot waiting.”

const fs = require('fs');
const path = require('path');
const zlib = require('zlib');

class BackupAutomation {
  constructor(options = {}) {
    this.backupDir = options.backupDir || path.join(__dirname, '../backups');
    this.targets = options.targets || [
      { name: 'patterns', path: path.join(__dirname, '../data/pattern-memory.json') },
      { name: 'trades', path: path.join(__dirname, '../logs/trades') },
      { name: 'config', path: path.join(__dirname, '../config.json') }
    ];
    this.interval = options.interval || 24 * 60 * 60 * 1000; // daily
    if (!fs.existsSync(this.backupDir)) fs.mkdirSync(this.backupDir, { recursive: true });
    this._start();
  }

  _start() {
    this._backup(); // initial
    this.timer = setInterval(() => this._backup(), this.interval);
  }

  _backup() {
    const stamp = new Date().toISOString().replace(/[:.]/g, '-');
    const snapDir = path.join(this.backupDir, `backup-${stamp}`);
    fs.mkdirSync(snapDir, { recursive: true });
    console.log(`🗂️ Creating backup: ${snapDir}`);

    this.targets.forEach(target => {
      const src = target.path;
      if (!fs.existsSync(src)) return;
      const stat = fs.statSync(src);
      const destPath = path.join(snapDir, `${target.name}${stat.isDirectory() ? '' : '.gz'}`);

      if (stat.isDirectory()) {
        // Copy entire directory recursively
        this._copyDir(src, path.join(snapDir, target.name));
      } else {
        // Compress file
        const buffer = fs.readFileSync(src);
        const gz = zlib.gzipSync(buffer);
        fs.writeFileSync(destPath, gz);
      }
    });

    console.log('✅ Backup complete');
    this._cleanupOldBackups();
  }

  _copyDir(src, dest) {
    if (!fs.existsSync(dest)) fs.mkdirSync(dest, { recursive: true });
    const items = fs.readdirSync(src);
    items.forEach(item => {
      const inPath = path.join(src, item);
      const outPath = path.join(dest, item);
      if (fs.statSync(inPath).isDirectory()) {
        this._copyDir(inPath, outPath);
      } else {
        fs.copyFileSync(inPath, outPath);
      }
    });
  }

  _cleanupOldBackups() {
    // Keep last 30 backups
    const entries = fs.readdirSync(this.backupDir).sort().reverse();
    const toDelete = entries.slice(30);
    toDelete.forEach(dir => {
      const fullPath = path.join(this.backupDir, dir);
      fs.rmSync(fullPath, { recursive: true, force: true });
      console.log(`🗑️ Deleted old backup: ${dir}`);
    });
  }

  stop() {
    clearInterval(this.timer);
  }
}

module.exports = BackupAutomation;
```

```javascript
// 📁 core/MonetizationManager.js
// “Show ’em the value—gate premium features until they pay the toll.”

const fs = require('fs');
const path = require('path');

class MonetizationManager {
  constructor(options = {}) {
    this.licensesFile = options.licensesFile || path.join(__dirname, '../licenses.json');
    this.licenseData = this._loadLicenses();
    this.currentLicense = null;
  }

  _loadLicenses() {
    if (!fs.existsSync(this.licensesFile)) {
      fs.writeFileSync(this.licensesFile, JSON.stringify({}));
    }
    return JSON.parse(fs.readFileSync(this.licensesFile, 'utf8'));
  }

  _saveLicenses() {
    fs.writeFileSync(this.licensesFile, JSON.stringify(this.licenseData, null, 2));
  }

  /**
   * Register a new license key for a user.
   * Expects { key: string, tier: 'free'|'pro'|'expert', expires: ISODateString }
   */
  registerLicense(license) {
    this.licenseData[license.key] = {
      tier: license.tier,
      expires: new Date(license.expires).toISOString()
    };
    this._saveLicenses();
    console.log(`🔑 Registered license ${license.key} as tier ${license.tier}`);
  }

  /**
   * Validate a license key (e.g., at startup).
   */
  validateKey(key) {
    const info = this.licenseData[key];
    if (!info) return false;
    if (new Date(info.expires) < new Date()) return false;
    this.currentLicense = info;
    console.log(`✅ License valid: tier ${info.tier}`);
    return true;
  }

  /**
   * Check if a feature is allowed for current tier.
   */
  isFeatureAllowed(featureName) {
    if (!this.currentLicense) return featureName === 'free';
    const { tier } = this.currentLicense;
    const featureTiers = {
      free: ['stochastic', 'basicPattern', 'mobileUI'],
      pro: ['stochastic', 'patternBoost', 'newsIntegration', 'monthlyReport', 'customAlerts'],
      expert: ['stochastic','patternBoost','newsIntegration','monthlyReport','customAlerts','quantumOpt','astroSignals']
    };
    return featureTiers[tier]?.includes(featureName) || false;
  }
}

module.exports = MonetizationManager;
```
